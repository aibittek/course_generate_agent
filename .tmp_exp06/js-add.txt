/* ================= QoS 实验台：兼容性检查 + 消息流模拟 ================= */
var SIM_N = 10;          // 模拟发布 10 条消息
var SIM_LATE_AT = 5;     // "订阅端晚到"时，从第几条消息起才算在线

function qosVals(){
  function v(id){ return document.getElementById(id).value; }
  return {
    pubRel: v('pubRel'), pubDur: v('pubDur'), pubDep: v('pubDep'),
    subRel: v('subRel'), subDur: v('subDur'), subDep: v('subDep'),
    late: document.getElementById('subLate').checked
  };
}
function relName(x){ return x === 'R' ? 'RELIABLE' : 'BEST_EFFORT'; }
function durName(x){ return x === 'T' ? 'TRANSIENT_LOCAL' : 'VOLATILE'; }
function depName(x){ return x === '0' ? 'KEEP_ALL' : 'KEEP_LAST(' + x + ')'; }

/* 兼容规则：订阅端"请求"不得高于发布端"提供" */
function compatReasons(c){
  var r = [];
  if(c.subRel === 'R' && c.pubRel === 'B')
    r.push('订阅端要求 RELIABLE，但发布端只提供 BEST_EFFORT —— 请求 > 提供，DDS 拒绝匹配');
  if(c.subDur === 'T' && c.pubDur === 'V')
    r.push('订阅端要求 TRANSIENT_LOCAL（要补发存量），但发布端 VOLATILE 不保留历史 —— 无法满足');
  if(c.subDep === '0' && c.pubDep !== '0')
    r.push('订阅端要求 KEEP_ALL 全部历史，但发布端只保留 KEEP_LAST 队列 —— 无法满足');
  return r;
}

function checkQos(){
  var c = qosVals();
  var reasons = compatReasons(c);
  var el = document.getElementById('qosResult');
  if(!el) return;
  if(reasons.length === 0){
    el.className = 'ok';
    el.textContent = '✔ 兼容 —— 两端 QoS 请求 ≤ 提供，DDS 将建立连接，消息可以正常分发';
  } else {
    el.className = 'bad';
    el.textContent = '✘ 不兼容 —— ' + reasons[0];
  }
}

/* 生成一次模拟的"逐条结果" */
function buildPlan(){
  var c = qosVals();
  var reasons = compatReasons(c);
  var pub = [], sub = [], i;
  for(i = 0; i < SIM_N; i++){ pub.push('sent'); sub.push('miss'); }
  var plan = { c: c, reasons: reasons, pub: pub, sub: sub,
               compatible: reasons.length === 0, got: 0,
               transLost: [], covered: [], missed: [], hold: [] };
  if(!plan.compatible) return plan;

  var start = c.late ? SIM_LATE_AT : 1;             // 第 start 条起订阅端在线

  /* ① 传输层：BEST_EFFORT 会真的丢包（固定丢第 3、7 条，便于对照） */
  for(i = 1; i <= SIM_N; i++){
    if(c.pubRel === 'B' && (i === 3 || i === 7)){ pub[i-1] = 'lost'; plan.transLost.push(i); }
  }
  /* ② 订阅端在线之前的数据：VOLATILE 一律错过 */
  for(i = 1; i <= SIM_N; i++){
    if(i < start){ sub[i-1] = 'miss'; plan.missed.push(i); continue; }
    sub[i-1] = (pub[i-1] === 'lost') ? 'lost' : 'recv';
  }
  /* ③ 晚到 + TRANSIENT_LOCAL：把加入前的"最后一条"补发过来 */
  if(c.late && c.pubDur === 'T' && start > 1){
    var k = start - 1;
    if(sub[k-1] === 'miss'){
      sub[k-1] = 'hold';
      plan.hold.push(k);
      plan.missed = plan.missed.filter(function(x){ return x !== k; });
    }
  }
  /* ④ 慢消费者 + 队列深度：只留得住 N 条，超出的被新消息覆盖 */
  var limit = (c.pubDep === '0') ? Infinity : parseInt(c.pubDep, 10);
  var vis = [];
  for(i = 1; i <= SIM_N; i++){ if(sub[i-1] === 'recv' || sub[i-1] === 'hold') vis.push(i); }
  if(vis.length > limit){
    var drop = vis.slice(0, vis.length - limit);
    for(i = 0; i < drop.length; i++){
      sub[drop[i]-1] = 'lost';
      plan.covered.push(drop[i]);
    }
    plan.hold = plan.hold.filter(function(x){ return drop.indexOf(x) === -1; });
  }
  plan.got = sub.filter(function(x){ return x === 'recv' || x === 'hold'; }).length;
  return plan;
}

function renderCells(){
  var p = document.getElementById('simPub'), s = document.getElementById('simSub');
  if(!p || !s) return;
  p.innerHTML = ''; s.innerHTML = '';
  for(var i = 1; i <= SIM_N; i++){
    var a = document.createElement('div'); a.className = 'cell'; a.textContent = i; p.appendChild(a);
    var b = document.createElement('div'); b.className = 'cell'; b.textContent = i; s.appendChild(b);
  }
}
function cellAt(box, i){ return document.getElementById(box).children[i-1]; }
function setCell(box, i, cls){ var el = cellAt(box, i); if(el) el.className = 'cell ' + cls; }

function resetSim(){
  renderCells();
  var log = document.getElementById('simLog');
  if(log) log.innerHTML = '点击「▶ 跑 10 条消息」开始模拟……';
  var vd = document.getElementById('simVerdict');
  if(vd){ vd.style.display = 'none'; vd.className = 'sim-verdict'; }
  checkQos();
}

function logHeader(c){
  return '<span class="d">$ ros2 topic echo /gas_sensor</span>\n'
    + '<span class="d">[发布端] ' + relName(c.pubRel) + ' · ' + durName(c.pubDur) + ' · ' + depName(c.pubDep) + '</span>\n'
    + '<span class="d">[订阅端] ' + relName(c.subRel) + ' · ' + durName(c.subDur) + ' · ' + depName(c.subDep)
    + (c.late ? ' · 第 ' + SIM_LATE_AT + ' 条之后才启动' : ' · 全程在线') + '</span>\n\n';
}
function rangeText(arr){
  if(!arr.length) return '';
  return arr.length > 3 ? ('第 ' + arr[0] + '–' + arr[arr.length-1] + ' 条') : ('第 ' + arr.join('、') + ' 条');
}

function simulateQos(){
  var plan = buildPlan();
  var c = plan.c;
  renderCells();
  var log = document.getElementById('simLog');
  var vd = document.getElementById('simVerdict');
  log.innerHTML = logHeader(c) + '<span class="d">模拟推进中……</span>';

  /* 逐条播放：先亮"发出"，再亮"收到/丢弃" */
  for(var i = 1; i <= SIM_N; i++){
    (function(i){
      setTimeout(function(){
        setCell('simPub', i, plan.pub[i-1] === 'lost' ? 'lost' : 'sent');
      }, i * 150);
      setTimeout(function(){
        setCell('simSub', i, plan.sub[i-1]);
      }, i * 150 + 90);
    })(i);
  }

  setTimeout(function(){
    var lines = [];
    if(!plan.compatible){
      lines.push('<span class="r">✘ 0/10 条送达 —— 两端在"发现阶段"就没匹配上</span>');
      lines.push('<span class="d">  └ ' + plan.reasons[0] + '</span>');
      lines.push('<span class="d">  订阅端不会报错，只是安静地什么都收不到 —— 这是 QoS 问题最难查的地方</span>');
      vd.style.display = 'block'; vd.className = 'sim-verdict bad';
      vd.textContent = '✘ DDS 拒绝匹配：订阅端一条都收不到';
    } else {
      lines.push('<span class="' + (plan.got === SIM_N ? 'g' : 'y') + '">'
        + (plan.got === SIM_N ? '✅ 送达 ' : '⚠️ 只送达 ') + plan.got + '/' + SIM_N + ' 条</span>');
      if(plan.transLost.length)
        lines.push('<span class="r">· 传输丢失 ' + plan.transLost.length + ' 条（' + rangeText(plan.transLost)
          + '）：BEST_EFFORT 尽力而为，网络一抖就丢，且不重传</span>');
      if(plan.missed.length)
        lines.push('<span class="r">· 错过 ' + plan.missed.length + ' 条（' + rangeText(plan.missed)
          + '）：订阅端当时还没上线，' + (c.pubDur === 'V'
            ? 'VOLATILE 不留存量，这些数据永远拿不回来'
            : 'TRANSIENT_LOCAL 只补"最后一条"，更早的照样拿不到') + '</span>');
      if(plan.hold.length)
        lines.push('<span class="y">✔ 补发存量 ' + plan.hold.length + ' 条（' + rangeText(plan.hold)
          + '）：TRANSIENT_LOCAL 把"最后一条"补给了晚到的订阅端</span>');
      if(plan.covered.length)
        lines.push('<span class="r">· 队列覆盖 ' + plan.covered.length + ' 条（' + rangeText(plan.covered)
          + '）：订阅端 0.8s 才处理一条，而 ' + depName(c.pubDep) + ' 留不住那么多，旧消息被新消息顶掉了</span>');
      if(!plan.transLost.length && !plan.missed.length && !plan.covered.length)
        lines.push('<span class="g">· 可靠传输 + 队列深度足够 + 订阅端全程在线 —— 一条都没丢</span>');
      vd.style.display = 'block';
      if(plan.got === SIM_N){ vd.className = 'sim-verdict ok'; vd.textContent = '✔ 10/10 条全部送达 —— 这份"合同"是健康的'; }
      else { vd.className = 'sim-verdict warn'; vd.textContent = '⚠️ 只送到 ' + plan.got + '/10 条 —— 看下面日志，找出是谁把消息弄丢了'; }
    }
    log.innerHTML = logHeader(c) + lines.join('\n');
  }, SIM_N * 150 + 320);
}

document.addEventListener('DOMContentLoaded', function(){
  ['pubRel','pubDur','pubDep','subRel','subDur','subDep'].forEach(function(id){
    var el = document.getElementById(id);
    if(el) el.addEventListener('change', checkQos);
  });
  var late = document.getElementById('subLate');
  if(late) late.addEventListener('change', checkQos);
  renderCells();
  checkQos();
});
