# talker.py — 最小发布者节点
import rclpy
from rclpy.node import Node
from std_msgs.msg import String

class Talker(Node):
    def __init__(self):
        super().__init__('talker')
        self.pub = self.create_publisher(String, 'chatter', 10)
        self.timer = self.create_timer(1.0, self.tick)

    def tick(self):
        msg = String()
        msg.data = 'Hello from C1!'
        self.pub.publish(msg)

rclpy.init()
rclpy.spin(Talker())

ROS2 底层通信依赖哪种机制？

A串口直连通信
BDDS 分布式数据分发
CHTTP 请求轮询
D共享内存互斥锁
✅ 正确答案是 B。ROS2 采用 DDS（Data Distribution Service）作为通信中间件，实现去中心化的节点发现与实时数据分发，无需 ROS1 中的 Master 节点。
